class Solution {
public:
    int climbStairs(int n) {

        int prev = 1, curr = 2, ans = 0;
        if(n == 1 || n == 2) return n;

        for(int i = 3; i <= n; i++){

            ans = curr + prev;
            prev = curr;
            curr = ans;

        }

        return ans;
        
    }
};
