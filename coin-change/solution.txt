class Solution {
public:

    int func(vector<int>& coins, int amount, int i, vector<vector<int>> &dp){

        if(amount == 0) return 0;
        if(i < 0 || amount < 0) return 1e9;

        if(dp[i][amount] != -1) return dp[i][amount];

        int take = 1 + func(coins, amount - coins[i], i, dp);
        int nottake = func(coins, amount, i - 1, dp);

        dp[i][amount] = min(take, nottake);
        return dp[i][amount];

    }

    int coinChange(vector<int>& coins, int amount) {

        int n = coins.size();
        vector<vector<int>> dp(coins.size(), vector<int>(amount+1, -1));

        int ans = func(coins, amount, n-1, dp);

        if(ans >= 1e9) return -1;
        return ans;
         
    }
};
