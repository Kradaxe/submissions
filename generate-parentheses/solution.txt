class Solution {
public:

    void generate(int n, vector<string> &ans, string s, stack<char> &st){

        if(n == 0 && st.size() == 0){
            ans.push_back(s);
            return;
        }

        if(n > 0){

            s += '(';
            st.push('(');

            generate(n-1, ans, s, st);

            st.pop();
            s.pop_back();       // ADD THIS
        }

        if(st.size()){

            s += ')';
            st.pop();

            generate(n, ans, s, st);

            st.push('(');
            s.pop_back();       // ADD THIS
        }
    }

    vector<string> generateParenthesis(int n) {

        vector<string> ans;
        stack<char> st;

        generate(n, ans, "", st);

        return ans;
    }
};
