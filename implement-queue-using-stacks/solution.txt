class MyQueue {

    stack<int> st1;
    stack<int> st2;
    int front;

public:
    MyQueue() {
        
    }
    
    void push(int x) {

        if(st1.size() == 0) front = x;
        st1.push(x);
        
    }
    
    int pop() {

        if (st2.size() == 0) {while(st1.size()){

            st2.push(st1.top());
            st1.pop();

        }}
        
        int ans = st2.top();
        st2.pop();

        return ans;
        
    }
    
    int peek() {

        if(st2.size()) return st2.top();
        return front;
        
    }
    
    bool empty() {

        return !(st1.size() + st2.size());
        
    }
};
