class Solution {
public:
    bool dfs(vector<vector<int>>& graph, vector<int> &v, int i, int prev){

       if(prev == -1) v[i] = 1;
       else if(prev == 1) v[i] = 2;
       else if(prev == 2) v[i] = 1;

       bool temp = true;

       for(int j = 0; j < graph[i].size(); j++){

            int next = v[graph[i][j]];
            if(next == -1) temp = dfs(graph, v, graph[i][j], v[i]);
            else if(next == v[i]) return false;
            else if(next != v[i]) continue;     

       }

       return temp;

    }

    bool isBipartite(vector<vector<int>>& graph) {

        int V = graph.size();
        vector<int> v(V, -1);

        bool ans;

        for(int i = 0; i < V; i++){     
            if(v[i] == -1) ans = dfs(graph, v, i, -1);
            if(ans == false) return false;
        }

        return true;
        
    }
};
