class Solution {
public:
    int longestPalindrome(string s) {

        unordered_map<char, int> mp;

        for(auto i : s) mp[i]++;

        int ans = 0;
        int flag = 0;

        for(auto it : mp){

            if(it.second % 2) flag = 1, ans += it.second-1;
            else ans += it.second;

        }
        
        return ans + flag;

    }
};
