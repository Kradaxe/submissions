class Solution {
public:
    vector<vector<int>> merge(vector<vector<int>>& intervals) {

        sort(intervals.begin(), intervals.end(), [](const vector<int> &a, const vector<int> &b){

            if(a[0] == b[0]) return a[1] < b[1];
            return a[0] < b[0];

        });

        if(intervals.size() == 1) return intervals;

        vector<vector<int>> ans;
        vector<int> curr{};
        
        for(auto it = 0; it < intervals.size() ; it++){

            if(!curr.size()) curr = intervals[it];
            else {

                if(curr[1] > intervals[it][0] && curr[1] > intervals[it][1]){

                    if(it == intervals.size() - 1) ans.push_back(curr);
                    continue;

                }
                else if(curr[1] < intervals[it][0]) {
                    
                    ans.push_back(curr), curr = intervals[it];
                    if(it == intervals.size() - 1) ans.push_back(curr);
                
                }
                else {
                    curr[1] = intervals[it][1];
                    if(it == intervals.size() - 1) ans.push_back(curr);
                }

            }

        }

        return ans;
        
    }
};
