class MinStack {

    stack<int> fir, sec;

public:
    MinStack() {
        
    }
    
    void push(int val) {

        fir.push(val);
        if(!sec.size() || getMin() >= val) sec.push(val);
        
    }
    
    void pop() {

        if(fir.top() == getMin()) sec.pop();
        fir.pop();
        
    }
    
    int top() {

        return fir.top();
        
    }
    
    int getMin() {

        return sec.top();
        
    }
};

/**
 * Your MinStack object will be instantiated and called as such:
 * MinStack* obj = new MinStack();
 * obj->push(val);
 * obj->pop();
 * int param_3 = obj->top();
 * int param_4 = obj->getMin();
 */
