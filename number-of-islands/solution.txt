class Solution {
public:

    void func(vector<vector<char>>& grid, vector<vector<int>>& vis, int i, int j, int n, int m){

        vis[i][j] = 1;

        vector<pair<int, int>> v = {{0,-1},{0,1},{1,0},{-1,0}};

        for(auto it : v){

            int x = it.first;
            int y = it.second;

            if(i + x < 0 || j + y < 0 || i + x >= n || j + y >= m || grid[i + x][j + y] == '0' ||vis[i + x][j + y] == 1) continue;

            func(grid, vis, i + x, j + y, n, m);

        }

        return;

    }

    int numIslands(vector<vector<char>>& grid) {

        int n = grid.size();
        int m = grid[0].size();

        int ans = 0;

        vector<vector<int>> vis(n, vector<int>(m, 0));

        for(int i = 0; i < n; i++){

            for(int j = 0; j < m; j++){

                if(grid[i][j] == '1' && vis[i][j] == 0) func(grid, vis, i, j, n, m), ans++;

            }

        }

        return ans;
        
    }
};
