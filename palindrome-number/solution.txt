class Solution {
public:
    bool isPalindrome(int x) {

        if(x == 0) return true;
        if(x < 0 || x % 10 ==0) return false;

        int comp = 0;

        while(comp < x){

            comp = 10*comp + x%10;
            x = x/10;

        }

        if(x == comp) return true;
        else if(x == comp/10) return true;

        return false;
        
    }
};
