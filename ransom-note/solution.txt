class Solution {
public:
    bool canConstruct(string ransomNote, string magazine) {

        if(ransomNote.size() > magazine.size()) return 0;

        vector<int> v(26, 0);

        for(int i = 0; i<magazine.size(); i++){

            if(i < ransomNote.size())v[ransomNote[i]-'a']++;
            v[magazine[i]-'a']--;

        }

        for(auto it : v){

            if(it > 0) return 0;

        }

        return 1;
        
    }
};
