class Solution {
public:
    void rotate(vector<vector<int>>& matrix) {

        int n = matrix.size();

        for(int i = n-1; i >= 0; i--){

            for(int j = i; j >= 0 ; j--) swap(matrix[n-1-i][j], matrix[n - j - 1][i]);

        }

        for(int i = 0; i < n/2; i++) swap(matrix[i], matrix[n-1-i]);
        
    }
};
