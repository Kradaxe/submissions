class Solution {
public:
    vector<int> sortedSquares(vector<int>& nums) {

        vector<int> ans;

        int i = 0, n = nums.size();
        while(i<n && nums[i] < 0) i++;
        int j = i-1;

        while(i < n && j >= 0){

            if(abs(nums[i]) > abs(nums[j])) ans.push_back(nums[j]*nums[j]), j--;
            else ans.push_back(nums[i]*nums[i]), i++;

        }

        if(i >= n){

            while(j >= 0){

                ans.push_back(nums[j]*nums[j]);
                j--;

            }

        }
        if(j < 0){

            while(i < n){

                ans.push_back(nums[i]*nums[i]);
                i++;

            }

        }

        return ans;
        
    }
};
