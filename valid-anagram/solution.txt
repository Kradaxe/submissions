class Solution {
public:
    bool isAnagram(string s, string t) {

        if(s.size() != t.size()) return 0;

        unordered_map<char, int> mp;

        for(auto i : s) mp[i]++;
        for(auto i : t) mp[i]--;

        for(auto it : mp){

            if (it.second > 0) return 0;

        }
        
        return 1;
    }
};
