class Solution {
public:

    bool isPal(string s){

        for(int i=0; i<s.size()/2; i++){

            if(s[i] != s[s.size()-1-i]) return 0;

        }

        return 1;

    }

    bool isPalindrome(string s) {

        string t = "";

        for(auto i : s){

            if((int(i)>=48 && int(i)<=57) || (int(i)>=65 && int(i)<=90) || (int(i)>=97 && int(i)<=122)) {
                
                i = tolower(i);
                t+=i;

            }

        }

        if(isPal(t)) return 1;
        return 0;
        
    }
};
